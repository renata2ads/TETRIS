# Tetris dla dzieci — plan wdrożenia

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Cel:** Przeglądarkowa gra typu Tetris (HTML + czysty JS) uruchamiana dwuklikiem na `index.html`, z logiką gry pokrytą testami w Node.

**Architektura:** Zwykłe skrypty `<script>` (bez modułów ES, bo nie działają z `file://`). `src/pieces.js` i `src/game.js` to czysta logika bez DOM — w przeglądarce dopisują się do `window.Tetris`, w Node eksportują przez `module.exports`. `src/view.js` rysuje planszę z 200 div-ów, obsługuje klawiaturę i zegar. Funkcje logiki przyjmują stan i zwracają nowy stan, nie zmieniając przekazanego.

**Technologie:** HTML5, CSS (grid), JavaScript ES2020 (CommonJS po stronie Node), Node 24 z wbudowanym `node:test` i `node:assert`. Żadnych pakietów npm.

**Specyfikacja:** [docs/superpowers/specs/2026-09-29-tetris-design.md](../specs/2026-09-29-tetris-design.md)

## Global Constraints

- Plansza: 10 kolumn × 20 wierszy, wiersz 0 = góra.
- Klocki: `I`, `O`, `T`, `S`, `Z`, `J`, `L`; każdy losowany z równym prawdopodobieństwem.
- Nowy klocek: pozycja obrotu 0, kolumna 3, najwyższy zajęty wiersz kształtu w wierszu 0 planszy.
- Punkty za wiersze zbite jednym zatrzymaniem: 0 → 0, 1 → 100, 2 → 300, 3 → 500, 4 → 800. Brak punktów za opadanie.
- Tempo stałe: 1 tick na 1000 ms. Brak poziomów.
- Obrót w prawo (0→1→2→3→0), bez wall kicka; kolizja = obrót się nie wykonuje.
- Klocek może wystawać ponad planszę; zatrzymanie z częścią ponad planszą = koniec gry.
- Statusy: `'playing'`, `'paused'`, `'over'`.
- Klawisze: ← / → ruch, Spacja obrót, ↓ jeden tick, ↑ nic, P pauza. `preventDefault` dla wszystkich tych klawiszy.
- Bez dźwięku, bez bibliotek, interfejs po polsku, plansza z elementów HTML (nie `<canvas>`).
- Pliki JS logiki muszą działać bez zmian i w przeglądarce (`file://`), i w Node (`require`).
- Nie używać modelu Sonnet 5.5 (ani dla subagentów, ani dla recenzentów).

## Review Focus

1. **Obrót klocka I zaraz po pojawieniu się** — jego pionowa postać wystaje ponad planszę; obrót ma się udać, a rysowanie nie może rzucić błędu. Test: Task 3.
2. **Zbicie wierszy, które nie sąsiadują ze sobą** (np. zbite 16, 18, 19, a 17 zostaje) — niezbity wiersz ma opaść dokładnie na dno. Test: Task 5.
3. **Klocek zatrzymany częściowo ponad planszą** — ma być koniec gry, a nie wyjątek przy zapisie do `board[-1]`. Test: Task 6.
4. **Przytrzymany klawisz P** — auto-repeat klawiatury nie może przełączać pauzy w kółko. Kod i ręczne sprawdzenie: Task 7.
5. **Wielokrotne „Zagraj ponownie” i spacja po kliknięciu przycisku** — tempo gry się nie zmienia (jeden zegar), a spacja obraca klocek zamiast ponownie „klikać” przycisk. Kod i ręczne sprawdzenie: Task 7.

```js
(function (exports) {
  'use strict';

  const TYPES = ['I', 'O', 'T', 'S', 'Z', 'J', 'L'];

  // Pozycja obrotu 0 każdego klocka: pola [wiersz, kolumna] w kwadracie size × size.
  const BASE = {
    I: { size: 4, cells: [[1, 0], [1, 1], [1, 2], [1, 3]] },
    T: { size: 3, cells: [[0, 1], [1, 0], [1, 1], [1, 2]] },
    S: { size: 3, cells: [[0, 1], [0, 2], [1, 0], [1, 1]] },
    Z: { size: 3, cells: [[0, 0], [0, 1], [1, 1], [1, 2]] },
    J: { size: 3, cells: [[0, 0], [1, 0], [1, 1], [1, 2]] },
    L: { size: 3, cells: [[0, 2], [1, 0], [1, 1], [1, 2]] },
  };
  const O_CELLS = [[0, 1], [0, 2], [1, 1], [1, 2]];

  function rotateClockwise(cells, size) {
    return cells.map(([r, c]) => [c, size - 1 - r]);
  }

  function buildRotations(type) {
    if (type === 'O') return [O_CELLS, O_CELLS, O_CELLS, O_CELLS];
    const { size, cells } = BASE[type];
    const rotations = [cells];
    for (let i = 1; i < 4; i++) rotations.push(rotateClockwise(rotations[i - 1], size));
    return rotations;
  }

  const SHAPES = {};
  TYPES.forEach((type) => {
    SHAPES[type] = buildRotations(type);
  });

  function getShape(type, rotation) {
    return SHAPES[type][rotation];
  }

  function spawnRow(type) {
    // "0 -" zamiast "-", żeby dla 0 nie powstało -0.
    return 0 - Math.min(...SHAPES[type][0].map(([r]) => r));
  }

  Object.assign(exports, { TYPES, getShape, spawnRow });
})(typeof module !== 'undefined' ? module.exports : (window.Tetris = window.Tetris || {}));
```

```js
function sequence(...types) {
  let i = 0;
  return () => {
    const type = types[Math.min(i, types.length - 1)];
    i++;
    return (TYPES.indexOf(type) + 0.5) / TYPES.length;
  };

function emptyBoard() {
  return Array.from({ length: 20 }, () => Array(10).fill(null));

// Pusta plansza z zajętymi polami [wiersz, kolumna].
function boardWith(cells) {
  const board = emptyBoard();
  cells.forEach(([r, c]) => {
    board[r][c] = 'X';
  });
  return board;

// Plansza, której dolne wiersze opisują napisy: '.' = puste, inny znak = zajęte.
function boardFrom(bottomRows) {
  const top = Array.from({ length: 20 - bottomRows.length }, () => Array(10).fill(null));
  const bottom = bottomRows.map((line) => [...line].map((ch) => (ch === '.' ? null : ch)));
  return [...top, ...bottom];
}
```

```js
(function (exports, pieces) {
  'use strict';

  const ROWS = 20;
  const COLS = 10;
  const SPAWN_COL = 3;

  function emptyRow() {
    return Array(COLS).fill(null);
  }

  function emptyBoard() {
    return Array.from({ length: ROWS }, emptyRow);
  }

  function randomType(random) {
    return pieces.TYPES[Math.floor(random() * pieces.TYPES.length)];
  }

  function newPiece(type) {
    return { type, rotation: 0, row: pieces.spawnRow(type), col: SPAWN_COL };
  }

  // Pola zajmowane przez klocek na planszy: [wiersz, kolumna].
  function pieceCells(piece) {
    return pieces.getShape(piece.type, piece.rotation).map(([r, c]) => [piece.row + r, piece.col + c]);
  }

  function createGame(random) {
    return { board: emptyBoard(), piece: newPiece(randomType(random)), score: 0, status: 'playing', random };
  }

  function getCells(state) {
    const cells = state.board.map((row) => row.slice());
    pieceCells(state.piece).forEach(([row, col]) => {
      cells[row][col] = state.piece.type;
    });
    return cells;
  }

  Object.assign(exports, { ROWS, COLS, createGame, getCells });
})(
  typeof module !== 'undefined' ? module.exports : window.Tetris,
  typeof module !== 'undefined' ? require('./pieces.js') : window.Tetris
);
```

```js
  // Kolizja: wyjście za lewą, prawą lub dolną krawędź albo zajęte pole. Nad planszą (wiersz < 0) wolno.
  function collides(board, piece) {
    return pieceCells(piece).some(
      ([row, col]) => col < 0 || col >= COLS || row >= ROWS || (row >= 0 && board[row][col] !== null)
    );
  }

  // Przy kolizji zwraca ten sam obiekt stanu — tick() rozpoznaje po tym, że klocek nie może opaść.
  function tryMove(state, change) {
    const piece = { ...state.piece, ...change(state.piece) };
    return collides(state.board, piece) ? state : { ...state, piece };
  }

  function moveLeft(state) {
    return tryMove(state, (piece) => ({ col: piece.col - 1 }));
  }

  function moveRight(state) {
    return tryMove(state, (piece) => ({ col: piece.col + 1 }));
  }

  function rotate(state) {
    return tryMove(state, (piece) => ({ rotation: (piece.rotation + 1) % 4 }));
  }
```

```js
(function () {
  'use strict';

  const { createGame, moveLeft, moveRight, rotate, tick, togglePause, getCells, COLS } = window.Tetris;
  const TICK_MS = 1000;

  const boardEl = document.getElementById('board');
  const scoreEl = document.getElementById('score');
  const finalScoreEl = document.getElementById('final-score');
  const pauseOverlay = document.getElementById('pause-overlay');
  const overOverlay = document.getElementById('over-overlay');
  const restartButton = document.getElementById('restart');

  // 200 pól tworzymy raz; potem zmieniamy tylko ich klasy.
  const cellEls = [];
  for (let i = 0; i < 200; i++) {
    const el = document.createElement('div');
    el.className = 'cell';
    boardEl.appendChild(el);
    cellEls.push(el);
  }

  let state = createGame(Math.random);

  function render() {
    getCells(state).forEach((row, r) => {
      row.forEach((type, c) => {
        cellEls[r * COLS + c].className = type ? `cell cell--${type}` : 'cell';
      });
    });
    scoreEl.textContent = state.score;
    finalScoreEl.textContent = state.score;
    pauseOverlay.hidden = state.status !== 'paused';
    overOverlay.hidden = state.status !== 'over';
  }

  function update(next) {
    if (next === state) return;
    state = next;
    render();
  }

  const ACTIONS = {
    ArrowLeft: moveLeft,
    ArrowRight: moveRight,
    ArrowDown: tick,
    ' ': rotate,
    p: togglePause,
    P: togglePause,
  };

  document.addEventListener('keydown', (event) => {
    if (event.key === 'ArrowUp') {
      event.preventDefault();
      return;
    }
    const action = ACTIONS[event.key];
    if (!action) return;
    event.preventDefault();
    // Przytrzymany P nie może przełączać pauzy w kółko.
    if (action === togglePause && event.repeat) return;
    update(action(state));
  });

  restartButton.addEventListener('click', () => {
    update(createGame(Math.random));
    // Bez fokusu na przycisku spacja obraca klocek, zamiast znowu go „klikać”.
    restartButton.blur();
  });

  // Jeden zegar na całe życie strony — restart podmienia tylko stan.
  setInterval(() => update(tick(state)), TICK_MS);

  render();
})();
```
